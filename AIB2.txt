def dfs(maze, x, y, path):
    if x < 0 or y < 0 or x >= len(maze) or y >= len(maze[0]):
        return False

    if maze[x][y] == 1:
        return False

    if maze[x][y] == 'G':
        path.append((x, y))
        return True

    maze[x][y] = 1
    path.append((x, y))

    if (dfs(maze, x+1, y, path) or
        dfs(maze, x, y+1, path) or
        dfs(maze, x-1, y, path) or
        dfs(maze, x, y-1, path)):
        return True

    path.pop()
    return False


maze = [
    ['S', 0, 1, 0],
    [0, 0, 1, 0],
    [1, 0, 0, 0],
    [1, 1, 1, 'G']
]

path = []
dfs(maze, 0, 0, path)

if path:
    print("Path:", path)
else:
    print("No path found")

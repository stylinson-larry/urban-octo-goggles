grid = [
    ['A', '.', '.'],
    ['#', '#', '.'],
    ['.', '.', 'G']
]
position = [0, 0]          # Internal state
goal = [2, 2]
def agent():
    global position
    while position != goal:
        r, c = position
        # Move right if possible
        if c + 1 < 3 and grid[r][c + 1] != '#':
            c += 1
        # Otherwise move down
        elif r + 1 < 3 and grid[r + 1][c] != '#':
            r += 1
        # Otherwise move up
        elif r - 1 >= 0 and grid[r - 1][c] != '#':
            r -= 1
        else:
            print("No path available")
            return
        position = [r, c]
        print("Current position:", position)

    print("Goal reached!")

agent()
